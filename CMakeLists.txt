cmake_minimum_required(VERSION 3.25)
project(ReworkPCControl VERSION 4.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Widgets Network ShaderTools)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(ReworkPCControl WIN32
    src/main.cpp src/ShutdownController.h src/ShutdownController.cpp
    src/ControlModel.h src/ControlModel.cpp src/HardwareTelemetry.h src/HardwareTelemetry.cpp
    src/AssistantLauncher.h src/AssistantLauncher.cpp src/app.rc)
target_sources(ReworkPCControl PRIVATE src/AstraLimits.h src/AstraLimits.cpp)
target_compile_definitions(ReworkPCControl PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(ReworkPCControl PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network dwmapi advapi32 user32 shell32 ole32 uuid)
file(GLOB APP_QML CONFIGURE_DEPENDS "qml/*.qml")
foreach(file IN LISTS APP_QML)
    get_filename_component(name "${file}" NAME)
    set_source_files_properties("${file}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
qt_add_qml_module(ReworkPCControl URI Rework VERSION 1.0 QML_FILES ${APP_QML})
file(GLOB APP_SHADERS CONFIGURE_DEPENDS "shaders/*.frag")
qt_add_shaders(ReworkPCControl "app_shaders" PREFIX "/shaders" BASE "shaders" FILES ${APP_SHADERS})
qt_add_resources(ReworkPCControl "assets" PREFIX "/" FILES rework.ico assets/diamond-dial.png)
include(CTest)
if(BUILD_TESTING)
    add_executable(model_tests tests/ModelTests.cpp src/ControlModel.cpp)
    target_include_directories(model_tests PRIVATE src)
    target_link_libraries(model_tests PRIVATE Qt6::Core)
    add_test(NAME countdown_safety COMMAND model_tests)
    add_executable(astra_tests tests/AstraLimitsTests.cpp src/AstraLimits.cpp src/AstraLimits.h)
    target_include_directories(astra_tests PRIVATE src)
    target_link_libraries(astra_tests PRIVATE Qt6::Core)
    add_test(NAME astra_limits_parsing COMMAND astra_tests)
endif()
